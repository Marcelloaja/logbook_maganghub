Panduan Google Drive Bersama (Untuk 33 Peserta)

Cukup 1 orang koordinator/admin yang memasang Google Apps Script ini sekali saja. Setelah itu, seluruh 33 peserta magang bisa menyimpan dan membagikan link dokumen secara otomatis tanpa perlu login akun Google masing-masing!

  1. Buka script.google.com (login dengan akun Google koordinator), lalu klik + New project.
  2. Hapus kode bawaan myFunction(), lalu tempel kode Google Apps Script di bawah ini:
// ============================================================
// GOOGLE APPS SCRIPT: LOGBOOK STORAGE HUB (33 PESERTA MAGANG)
// Otomatis membaca & sinkronisasi seluruh folder fisik Google Drive
// ============================================================

function doGet(e) {
  try {
    return handleGetSyncData();
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      success: false,
      error: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function handleGetSyncData() {
  var rootFolderName = "Logbook Magang Balai Bahasa Jatim";
  var rootFolders = DriveApp.getFoldersByName(rootFolderName);
  var rootFolder = rootFolders.hasNext() ? rootFolders.next() : DriveApp.createFolder(rootFolderName);
  rootFolder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

  var folders = [];
  var files = [];

  // 1. Baca LANGSUNG seluruh subfolder fisik yang ada di Google Drive
  var subFolders = rootFolder.getFolders();
  while (subFolders.hasNext()) {
    var sub = subFolders.next();
    var fName = sub.getName();
    if (fName.toLowerCase() === "umum") continue;
    var fId = sub.getId();
    folders.push({
      id: fId,
      name: fName,
      createdAt: Utilities.formatDate(sub.getDateCreated(), "Asia/Jakarta", "yyyy-MM-dd")
    });

    // Baca file-file di dalam subfolder ini
    var subFiles = sub.getFiles();
    while (subFiles.hasNext()) {
      var file = subFiles.next();
      var fExt = file.getName().toLowerCase().indexOf(".docx") !== -1 ? "docx" : "pdf";
      files.push({
        id: file.getId(),
        folderId: fId,
        folderName: fName,
        name: file.getName(),
        fileName: file.getName(),
        format: fExt,
        url: "https://drive.google.com/file/d/" + file.getId() + "/view?usp=sharing",
        date: Utilities.formatDate(file.getDateCreated(), "Asia/Jakarta", "yyyy-MM-dd HH:mm"),
        uploadDate: Utilities.formatDate(file.getDateCreated(), "Asia/Jakarta", "yyyy-MM-dd HH:mm"),
        size: file.getSize()
      });
    }
  }

  // 2. Baca file yang ada di root folder (jika ada)
  var rootFiles = rootFolder.getFiles();
  while (rootFiles.hasNext()) {
    var rFile = rootFiles.next();
    var rExt = rFile.getName().toLowerCase().indexOf(".docx") !== -1 ? "docx" : "pdf";
    files.push({
      id: rFile.getId(),
      folderId: rootFolder.getId(),
      folderName: "Umum",
      name: rFile.getName(),
      fileName: rFile.getName(),
      format: rExt,
      url: "https://drive.google.com/file/d/" + rFile.getId() + "/view?usp=sharing",
      date: Utilities.formatDate(rFile.getDateCreated(), "Asia/Jakarta", "yyyy-MM-dd HH:mm"),
      uploadDate: Utilities.formatDate(rFile.getDateCreated(), "Asia/Jakarta", "yyyy-MM-dd HH:mm"),
      size: rFile.getSize()
    });
  }

  // Urutkan nama folder A-Z
  folders.sort(function(a, b) {
    return a.name.localeCompare(b.name);
  });

  // 3. Baca link dokumentasi kegiatan (dokumentasi_links.json)
  var drivelinks = getDriveLinksFromCloud(rootFolder);

  return ContentService.createTextOutput(JSON.stringify({
    success: true,
    folders: folders,
    files: files,
    links: drivelinks
  })).setMimeType(ContentService.MimeType.JSON);
}

function getDriveLinksFromCloud(rootFolder) {
  try {
    var files = rootFolder.getFilesByName("dokumentasi_links.json");
    if (files.hasNext()) {
      var content = files.next().getBlob().getDataAsString();
      return JSON.parse(content || "[]");
    }
  } catch (err) {
    Logger.log("Read links error: " + err);
  }
  return [];
}

function saveDriveLinksToCloud(rootFolder, links) {
  try {
    var content = JSON.stringify(links || []);
    var files = rootFolder.getFilesByName("dokumentasi_links.json");
    if (files.hasNext()) {
      var file = files.next();
      file.setContent(content);
    } else {
      var newFile = rootFolder.createFile("dokumentasi_links.json", content, MimeType.PLAIN_TEXT);
      newFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    }
  } catch (err) {
    Logger.log("Save links error: " + err);
  }
}

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);

    // ACTION 0: SINKRONISASI DATA FOLDER & BERKAS
    if (data.action === "getSyncData") {
      return handleGetSyncData();
    }

    var rootFolderName = "Logbook Magang Balai Bahasa Jatim";
    var rootFolders = DriveApp.getFoldersByName(rootFolderName);
    var rootFolder = rootFolders.hasNext() ? rootFolders.next() : DriveApp.createFolder(rootFolderName);
    rootFolder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

    // ACTION 1: BUAT FOLDER BARU
    if (data.action === "createFolder") {
      var folderName = (data.folderName || "").trim();
      if (!folderName) throw new Error("Nama folder tidak boleh kosong");
      
      var existingSub = rootFolder.getFoldersByName(folderName);
      var driveSub = existingSub.hasNext() ? existingSub.next() : rootFolder.createFolder(folderName);
      driveSub.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

      return ContentService.createTextOutput(JSON.stringify({
        success: true,
        folder: {
          id: driveSub.getId(),
          name: folderName,
          createdAt: Utilities.formatDate(new Date(), "Asia/Jakarta", "yyyy-MM-dd")
        }
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // ACTION 2: HAPUS FILE DARI GOOGLE DRIVE
    if (data.action === "deleteFile") {
      var fileId = data.fileId;
      var fileDeleted = false;
      if (fileId && fileId.indexOf("file-") !== 0) {
        try {
          DriveApp.getFileById(fileId).setTrashed(true);
          fileDeleted = true;
        } catch (e) {}
      }
      if (!fileDeleted && data.fileName && data.folderName) {
        var subFolders = rootFolder.getFoldersByName(data.folderName);
        if (subFolders.hasNext()) {
          var targetFolder = subFolders.next();
          var matchFiles = targetFolder.getFilesByName(data.fileName);
          while (matchFiles.hasNext()) {
            matchFiles.next().setTrashed(true);
            fileDeleted = true;
          }
        }
      }
      return ContentService.createTextOutput(JSON.stringify({
        success: true,
        deleted: fileDeleted
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // ACTION 3: HAPUS FOLDER BESERTA ISINYA DARI GOOGLE DRIVE
    if (data.action === "deleteFolder") {
      var folderId = data.folderId;
      var folderName = (data.folderName || "").trim();
      var folderDeleted = false;

      if (folderId && folderId.indexOf("f-") !== 0) {
        try {
          DriveApp.getFolderById(folderId).setTrashed(true);
          folderDeleted = true;
        } catch (e) {}
      }

      if (!folderDeleted && folderName) {
        var matchFolders = rootFolder.getFoldersByName(folderName);
        while (matchFolders.hasNext()) {
          matchFolders.next().setTrashed(true);
          folderDeleted = true;
        }
      }

      return ContentService.createTextOutput(JSON.stringify({
        success: true,
        deleted: folderDeleted
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // ACTION 4: GET DOKUMENTASI DRIVE LINKS
    if (data.action === "getDriveLinks") {
      return ContentService.createTextOutput(JSON.stringify({
        success: true,
        links: getDriveLinksFromCloud(rootFolder)
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // ACTION 5: SAVE / UPDATE DOKUMENTASI DRIVE LINK (SINGLE / BATCH)
    if (data.action === "saveDriveLink" || data.action === "saveDriveLinksBatch") {
      var currentLinks = getDriveLinksFromCloud(rootFolder);
      var itemsToSave = data.action === "saveDriveLinksBatch" ? (data.linkItems || []) : [data.linkItem];
      for (var b = 0; b < itemsToSave.length; b++) {
        var linkItem = itemsToSave[b];
        if (!linkItem || !linkItem.id) continue;
        var foundIdx = -1;
        for (var k = 0; k < currentLinks.length; k++) {
          if (currentLinks[k].id === linkItem.id || (linkItem.url && currentLinks[k].url === linkItem.url)) {
            foundIdx = k;
            break;
          }
        }
        if (foundIdx !== -1) {
          currentLinks[foundIdx] = linkItem;
        } else {
          currentLinks.unshift(linkItem);
        }
      }
      saveDriveLinksToCloud(rootFolder, currentLinks);
      return ContentService.createTextOutput(JSON.stringify({
        success: true,
        links: currentLinks
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // ACTION 6: DELETE DOKUMENTASI DRIVE LINK
    if (data.action === "deleteDriveLink") {
      var linkIdToDelete = data.linkId;
      var currentLinks2 = getDriveLinksFromCloud(rootFolder);
      var filteredLinks = [];
      for (var m = 0; m < currentLinks2.length; m++) {
        if (currentLinks2[m].id !== linkIdToDelete) {
          filteredLinks.push(currentLinks2[m]);
        }
      }
      saveDriveLinksToCloud(rootFolder, filteredLinks);
      return ContentService.createTextOutput(JSON.stringify({
        success: true,
        links: filteredLinks
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // ACTION 7: SIMPAN / UPLOAD DOKUMEN LOGBOOK
    var targetFolderName = (data.folderName || "").trim();
    var targetFolder = rootFolder;
    if (targetFolderName && targetFolderName.toLowerCase() !== "umum") {
      var subFolders = rootFolder.getFoldersByName(targetFolderName);
      targetFolder = subFolders.hasNext() ? subFolders.next() : rootFolder.createFolder(targetFolderName);
      targetFolder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    }

    var decoded = Utilities.base64Decode(data.base64);
    var blob = Utilities.newBlob(decoded, data.mimeType, data.fileName);
    var file = targetFolder.createFile(blob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

    var fileUrl = "https://drive.google.com/file/d/" + file.getId() + "/view?usp=sharing";
    var docExt = file.getName().toLowerCase().indexOf(".docx") !== -1 ? "docx" : "pdf";
    var fileObj = {
      id: file.getId(),
      folderId: targetFolder.getId(),
      folderName: targetFolderName,
      name: file.getName(),
      fileName: file.getName(),
      format: docExt,
      url: fileUrl,
      date: Utilities.formatDate(new Date(), "Asia/Jakarta", "yyyy-MM-dd HH:mm"),
      uploadDate: Utilities.formatDate(new Date(), "Asia/Jakarta", "yyyy-MM-dd HH:mm"),
      size: file.getSize()
    };

    return ContentService.createTextOutput(JSON.stringify({
      success: true,
      url: fileUrl,
      folderUrl: targetFolder.getUrl(),
      file: fileObj
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      success: false,
      error: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}
